ประเภทหัตถการหลัก (Main Services) บริการหลักคลินิก

กำหนดประเภทหัตถการหลัก เช่น จองนวด, จองฟื้นฟูหลังคลอด ราคา รอบเวลา และสิทธิ์การมองเห็น

ชื่อหัตถการหลัก ไอคอน รายละเอียด ราคา/ค่าบริการ ระยะเวลา สิทธิ์แสดงผล สถานะ จัดการ

รายการบริการเสริม (Extra Services) หัตถการเพิ่มเติม

เพิ่มบริการเสริมใหม่ กำหนดราคา สัดส่วนส่วนแบ่งรายได้ และตั้งค่าการครองคิว 2 รอบเวลา

ชื่อบริการ ชื่อย่อ (แท็ก) ราคา (บาท) ส่วนแบ่ง (ผู้ช่วย / รพ.) การแสดงผล ครองคิว 2 รอบ สถานะ จัดการ

เชื่อมต่อฐานข้อมูล Supabase (PostgreSQL + Realtime)

ฐานข้อมูลคลาวด์ความเร็วสูง ซิงค์ข้อมูลข้ามอุปกรณ์สดแบบ Realtime ตลอด 24 ชม.

ยังไม่ได้เชื่อมต่อ (โหมดออฟไลน์)
คำสั่งสร้างตารางบน Supabase SQL Editor (supabase-schema.sql)

นำคำสั่ง SQL ด้านล่างนี้ไปวางใน Supabase Dashboard -> SQL Editor -> Run เพื่อสร้างตาราง `appointments`, `assistants`, `extra_services`, `slot_configs` พร้อมเปิด Realtime ทันที 

-- 1. Appointments Table
CREATE TABLE IF NOT EXISTS public.appointments (
    id TEXT PRIMARY KEY,
    patient_name TEXT NOT NULL,
    citizen_or_hn TEXT,
    phone TEXT NOT NULL,
    medical_scheme TEXT DEFAULT 'บัตรทอง',
    book_date DATE NOT NULL,
    time_slot TEXT NOT NULL,
    main_service TEXT NOT NULL,
    extra_services JSONB DEFAULT '[]'::jsonb,
    assistant_id TEXT,
    assistant_nick TEXT,
    status TEXT DEFAULT '⚪ ว่าง',
    slots_occupied JSONB DEFAULT '[]'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Assistants Table
CREATE TABLE IF NOT EXISTS public.assistants (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    nickname TEXT NOT NULL,
    active BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2.5 Main Services Table (ประเภทหัตถการหลัก)
CREATE TABLE IF NOT EXISTS public.main_services (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    title TEXT,
    icon TEXT DEFAULT '💆‍♂️',
    "desc" TEXT,
    price NUMERIC NOT NULL DEFAULT 0,
    price_label TEXT,
    duration_slots NUMERIC DEFAULT 1,
    duration_min NUMERIC DEFAULT 60,
    target_audience TEXT DEFAULT 'all',
    active BOOLEAN DEFAULT true,
    color TEXT DEFAULT 'bg-herbal-100 text-herbal-800 border-herbal-200',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. Extra Services Table
CREATE TABLE IF NOT EXISTS public.extra_services (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    tag TEXT NOT NULL,
    price NUMERIC NOT NULL DEFAULT 0,
    asst_percent NUMERIC DEFAULT 60,
    hospital_percent NUMERIC DEFAULT 40,
    share60 NUMERIC NOT NULL DEFAULT 0,
    target_audience TEXT DEFAULT 'all',
    two_slots BOOLEAN DEFAULT false,
    active BOOLEAN DEFAULT true,
    color TEXT DEFAULT 'bg-slate-100 text-slate-800 border border-slate-200',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 4. Slot Configs Table
CREATE TABLE IF NOT EXISTS public.slot_configs (
    id TEXT PRIMARY KEY,
    scope TEXT NOT NULL,
    config_key TEXT NOT NULL,
    slots_json JSONB NOT NULL DEFAULT '{}'::jsonb,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 5. Users Table
CREATE TABLE IF NOT EXISTS public.users (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    phone TEXT UNIQUE NOT NULL,
    email TEXT UNIQUE,
    auth_user_id UUID UNIQUE,
    role TEXT NOT NULL DEFAULT 'user',
    active BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 6. Audit Logs Table
CREATE TABLE IF NOT EXISTS public.audit_logs (
    id TEXT PRIMARY KEY,
    user_name TEXT NOT NULL,
    user_role TEXT NOT NULL,
    action_type TEXT NOT NULL,
    description TEXT NOT NULL,
    details JSONB DEFAULT '{}'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);
